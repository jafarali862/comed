public function view(string $id)
    { 
       $insuranceCustomer = InsuranceCustomer::all();
    $companies = InsuranceCompany::where('status', '!=', 0)->get();
    $users = User::whereNotIn('role', [1, 2])
                 ->where('status', '!=', 0)
                 ->get();

    // Fetch single customer with related data
    $customer = InsuranceCustomer::with([
            'company:id,name',
            'insuranceCase:id,customer_id,case_status,assigned_status,case_details,insurance_type',
            'caseAssignments:id,customer_id,type,date,other,executive_driver,executive_garage,executive_spot,executive_meeting,executive_accident_person,created_at',
            'caseAssignments.driver:id,name',
            'caseAssignments.garage:id,name',
            'caseAssignments.spot:id,name',
            'caseAssignments.meeting:id,name',
            'caseAssignments.accidentPerson:id,name',
        ])
        ->findOrFail($id);

    // Flatten data to look like old query
    $customers = (object)[
        'id' => $customer->id,
        'name' => $customer->name,
        'phone' => $customer->phone,
        'crime_number' => $customer->crime_number,
        'police_station' => $customer->police_station,
        'case_status' => $customer->insuranceCase->case_status ?? null,
        'assigned_status' => $customer->insuranceCase->assigned_status ?? null,
        'case_details' => $customer->insuranceCase->case_details ?? null,
        'ins_type' => $customer->insuranceCase->insurance_type ?? null,
        'custname' => $customer->company->name ?? null,
        'case_type' => $customer->caseAssignments->type ?? null,
        'case_date' => $customer->caseAssignments->date ?? null,
        'case_other' => $customer->caseAssignments->other ?? null,
        'driver_id' => $customer->caseAssignments->driver->id ?? null,
        'garage_id' => $customer->caseAssignments->garage->id ?? null,
        'spot_id' => $customer->caseAssignments->spot->id ?? null,
        'meeting_id' => $customer->caseAssignments->meeting->id ?? null,
        'accident_id' => $customer->caseAssignments->accidentPerson->id ?? null,
        'driver_name' => $customer->caseAssignments->driver->name ?? null,
        'garage_name' => $customer->caseAssignments->garage->name ?? null,
        'spot_name' => $customer->caseAssignments->spot->name ?? null,
        'meeting_name' => $customer->caseAssignments->meeting->name ?? null,
        'accident_name' => $customer->caseAssignments->accidentPerson->name ?? null,
        'created_at' => $customer->caseAssignments->created_at ?? $customer->created_at,
    ];

    return view('dashboard.case.view', compact('customers', 'companies', 'id', 'users', 'insuranceCustomer'));  
    }
